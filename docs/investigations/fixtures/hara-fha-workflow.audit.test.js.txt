jest.mock('./aiAnalysisSTPA', () => ({fetchLLMResponse:jest.fn()}));
jest.mock('./backendConfig', () => ({backendURL:'https://example.test',buildAuthOpts:()=>({}),buildAIAuthOpts:()=>({})}));
const {fetchLLMResponse}=require('./aiAnalysisSTPA');
const {generateHaraAnalysisSheets,generateFhaAnalysisSheets}=require('./aiAnalysisHaraFha');
const {buildCodeArchitectureHazardInput}=require('../features/code-architecture-hazard-analysis/codeArchitectureHazardUtils');
const {importHazardSoftwareRequirements}=require('../features/code-architecture-assurance/artifactAI');
const source={from:'Validate brake command',action:'Apply brake torque',to:'Brake controller',traceId:'trace-1',rowRef:'1',hazardAnalysisEligibility:'Include'};
const contexts=[{id:'urban',scenario:'Urban driving',mode:'Automatic'},{id:'service',scenario:'Service depot',mode:'Maintenance'}];
const response=[1,2].map(i=>({id:`FD-${i}`,itemFunction:'Validate brake command',functionName:'Validate brake command',malfunction:'Brake torque exceeds requested torque',hazard:'Brake controller applies excess torque on a wet road and vehicle loses traction',operationalSituation:'Urban wet-road braking',hazardousEvent:'Excess brake torque locks the wheel during a turn',harm:'Vehicle occupants injured',severity:'S3',exposure:'E4',controllability:'C3',asil:'ASIL D',safetyGoal:'The system shall limit brake torque to the validated request.',functionalDegradationOrLoss:'Excess brake torque',mishap:'Vehicle collision',effect:'Wheel locks during a turn',severityCategory:'I - Catastrophic',softwareControlCategory:'1 - Autonomous',causalFactors:'Brake command limit omitted',controls:'Validate brake torque request',safetyRequirement:'The software shall validate the brake torque request.',verification:'Test brake torque limits',rationale:'Brake controller controls wheel torque',safetySignificant:'Yes',safetySignificanceRationale:'Excess brake torque can cause occupant injury',proposedSafetyAssessment:'Safety Related'}));
beforeEach(()=>{fetchLLMResponse.mockReset();jest.spyOn(console,'warn').mockImplementation(()=>{});});
afterEach(()=>jest.restoreAllMocks());
test.each([['HARA',generateHaraAnalysisSheets],['FHA',generateFhaAnalysisSheets]])('%s real generator loses row context but successful requirements have method/run provenance', async(method,generate)=>{
 const input=buildCodeArchitectureHazardInput({cbaRows:[source],method,projectId:'p',repoMeta:{repoId:'repo'},operationalContexts:contexts});
 expect(input.tableRows).toHaveLength(2);
 fetchLLMResponse.mockResolvedValue(JSON.stringify(response));
 const result=await generate({sheets:input.sheets,setFolders:jest.fn(),currentFolder:'cba',operationalContext:input.operationalContext});
 const headers=result.Summary[0];
 expect(headers).not.toContain('Operational Context ID');
 const requirements=importHazardSoftwareRequirements({hazardAnalysis:{id:'run',hazardMethod:method,generatedSheets:result}});
 expect(requirements.length).toBeGreaterThan(0);
 expect(requirements[0]).toMatchObject({hazardAnalysisMethod:method,hazardAnalysisRunId:'run',source:'hazard-derived'});
 console.log(method, JSON.stringify({summaryRows:result.Summary.length-1,requirements:requirements.length,verification:requirements[0].linkedVerification}));
});
test.each([['HARA',generateHaraAnalysisSheets],['FHA',generateFhaAnalysisSheets]])('%s returns populated fallback hazards after all provider failures',async(method,generate)=>{
 const input=buildCodeArchitectureHazardInput({cbaRows:[source],method});
 fetchLLMResponse.mockRejectedValue(new Error('provider unavailable'));
 const result=await generate({sheets:input.sheets,setFolders:jest.fn(),currentFolder:'cba'});
 const h=result.Summary[0], row=result.Summary[1];
 expect(row[h.indexOf('Hazard')]).toMatch(/Needs review/i);
 expect(String(row[h.indexOf('Hazard')]).trim()).not.toBe('');
 expect(importHazardSoftwareRequirements({hazardAnalysis:{generatedSheets:result}})).toEqual([]);
});
test.each([['HARA',generateHaraAnalysisSheets],['FHA',generateFhaAnalysisSheets]])('%s missing generation can become a placeholder software requirement when the classifier returns Yes',async(method,generate)=>{
 const input=buildCodeArchitectureHazardInput({cbaRows:[source],method});
 fetchLLMResponse.mockImplementation(async prompt=>prompt.includes('You are reviewing generated')
  ? JSON.stringify([{id:'FD-1',safetySignificant:'Yes',safetySignificanceRationale:'Potential occupant harm',proposedSafetyAssessment:'Safety'}]) : '[]');
 const result=await generate({sheets:input.sheets,setFolders:jest.fn(),currentFolder:'cba'});
 const requirements=importHazardSoftwareRequirements({hazardAnalysis:{id:'run',hazardMethod:method,generatedSheets:result}});
 expect(requirements).toHaveLength(1);
 expect(requirements[0].requirementText).toMatch(/needs review/i);
});
