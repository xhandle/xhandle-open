test('CBA section selection omits separately authored safety classification policy',()=>{
 const context=buildEffectiveOrganizationProfileContext({profile:createOrganizationProfileRecord({enabled:true,markdown:prefix+'# Safety Philosophy\nPHILOSOPHY\n# Safety Significance Classification\nCLASSIFICATION-RULE'}),sectionNames:selected});
 expect(context).toContain('PHILOSOPHY');expect(context).not.toContain('CLASSIFICATION-RULE');
});
test('long CBA profile loses risk and engineering sections during prioritization',()=>{
 const context=buildEffectiveOrganizationProfileContext({profile:createOrganizationProfileRecord({enabled:true,markdown:prefix+'# Safety Philosophy\n'+ 'x'.repeat(19000)+'\n# Risk Classification\nRISK-RULE\n# Engineering Rules\nENGINEERING-RULE'}),sectionNames:selected});
 expect(context).toContain('RISK-RULE');
 const prompt=formatGovernedHazardPromptContext({organizationContext:context});
 expect(prompt).not.toContain('RISK-RULE');expect(prompt).not.toContain('ENGINEERING-RULE');
});
