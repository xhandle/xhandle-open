global.IS_REACT_ACT_ENVIRONMENT=true;
jest.mock('lucide-react',()=>{const Icon=()=> <span/>;return new Proxy({},{get:()=>Icon});});
const React=require('react');
const {act}=React;
const {createRoot}=require('react-dom/client');
const Panel=require('./CodeArchitectureHazardPanel').default;
const empty=[];
const makeRun=id=>({id,projectId:'p',repoId:'r',hazardMethod:'STPA',generatedSheets:{Summary:[
 ['Raw Analysis Row ID','Function (From)','Control Action','Guide Phrase Applicable','Operational Context ID'],
 ...Array.from({length:602},(_,i)=>[id+'-'+i,'Function '+Math.floor(i/7),'Command','Yes','ctx'])
]}});
let host,root;
beforeEach(()=>{jest.useFakeTimers();host=document.createElement('div');document.body.appendChild(host);root=createRoot(host);});
afterEach(()=>{act(()=>root.unmount());host.remove();jest.useRealTimers();});
const render=props=>act(()=>root.render(<Panel method="STPA" cbaRows={empty} operationalContexts={empty} {...props}/>));
const settle=()=>{act(()=>jest.runAllTimers());act(()=>jest.runAllTimers());};
it('preserves the threshold table and group state in StrictMode',()=>{
 const full=makeRun('strict-boundary');
 const small={...full,generatedSheets:{Summary:full.generatedSheets.Summary.slice(0,501)}};
 const strict=run=>act(()=>root.render(<React.StrictMode><Panel method="STPA" cbaRows={empty} operationalContexts={empty} latestRun={run} isRunning/></React.StrictMode>));
 strict(small);settle();
 act(()=>[...host.querySelectorAll('button')].find(b=>b.textContent.includes('Collapse all groups')).click());
 const table=host.querySelector('table');
 for(const run of [full,small,full]) {
  strict(run);settle();
  expect(host.querySelector('table')).toBe(table);
  expect(host.querySelectorAll('tr[id^="cba-hazard-row-"]')).toHaveLength(0);
  expect(host.textContent).toContain('Stop analysis');
 }
});
it('does not let interrupted preparation overwrite a replacement small run',()=>{
 const first=makeRun('first');
 const replacement=makeRun('replacement');
 replacement.projectId='replacement-project';
 const small={...replacement,generatedSheets:{Summary:replacement.generatedSheets.Summary.slice(0,501)}};
 render({latestRun:first});
 act(()=>jest.advanceTimersToNextTimer());
 act(()=>jest.advanceTimersToNextTimer());
 render({latestRun:small});settle();
 const table=host.querySelector('table');
 expect(host.querySelector('[data-window-key="row:replacement-0"]')).not.toBeNull();
 expect(host.querySelector('[data-window-key="row:first-0"]')).toBeNull();
 render({latestRun:replacement});settle();
 expect(host.querySelector('table')).toBe(table);
 expect(host.querySelector('[data-window-key="row:first-0"]')).toBeNull();
 render({latestRun:first});settle();
 expect(host.querySelector('[data-window-key="row:first-0"]')).not.toBeNull();
 expect(host.querySelector('[data-window-key="row:replacement-0"]')).toBeNull();
});
